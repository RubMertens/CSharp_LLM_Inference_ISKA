public static float operator *(Vector a, Vector b)
{
    if (a.Length != b.Length)
        throw new InvalidOperationException("Incompatible vector dimensions for dot product.");

    float sum = 0;
    for (var i = 0; i < a.Length; i++)
    {
        sum += a.Data[i] * b.Data[i];
    }
    return sum;
}
